using Alis.Reactive.Elements;
using Alis.Reactive.Pipelines;
using Alis.Reactive.Requests;

namespace Intake;

public static class HidesReplyPlant
{
    public static void ANestedReplyNamedAsTheReplyAroundIt(Pipeline<IntakeView> pipeline, RequestForm<IntakeRequest> intake)
        => pipeline.Post<IntakeView, IntakeRequest>("/Wards/{ward}/Intake", request => request.Form(intake))
            .OnSuccess<IntakeCreated>((created, pipeline) => pipeline
                .Post<IntakeView, IntakeRequest>("/Wards/{ward}/Intake", request => request.Form(intake))
                .OnSuccess<IntakeCreated>((created, pipeline) => pipeline.Element("admitted").Text.Set(created.Request.Read(r => r.Name))));
}
