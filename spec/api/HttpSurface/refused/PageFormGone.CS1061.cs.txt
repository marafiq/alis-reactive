using Alis.Reactive.Elements;
using Alis.Reactive.Pipelines;
using Alis.Reactive.Requests;

namespace Intake;

public static class PageFormGone
{
    public static void Refused(Pipeline<IntakeView> pipeline)
        => pipeline.Post("/Wards/Census/Refresh")
            .OnError<ValidationProblem>(400, (refused, shown) => shown.Form("intake").ShowErrors(refused.Body));
}
