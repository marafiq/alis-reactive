using Alis.Reactive.Elements;
using Alis.Reactive.Pipelines;
using Alis.Reactive.Requests;

namespace Intake;

public static class InputsInJson
{
    public static void Refused(Pipeline<IntakeView> pipeline, RequestContainer<IntakeRequest> intake)
        => pipeline.Post<IntakeView, IntakeRequest>("/Wards/{ward}/Intake", request => request
            .Json(body => body.Inputs(intake)));
}
