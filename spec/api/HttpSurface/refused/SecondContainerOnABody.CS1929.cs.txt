using Alis.Reactive.Elements;
using Alis.Reactive.Pipelines;
using Alis.Reactive.Requests;

namespace Intake;

public static class SecondContainerOnABody
{
    public static void Refused(Pipeline<IntakeView> pipeline, RequestForm<IntakeRequest> intake, RequestContainer<IntakeRequest> details)
        => pipeline.Post<IntakeView, IntakeRequest>("/Wards/{ward}/Intake", request => request.Form(intake).Inputs(details));
}
