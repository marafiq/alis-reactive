using System;
using Alis.Reactive.Elements;
using Alis.Reactive.Pipelines;
using Alis.Reactive.Requests;

namespace Intake;

public static class RetryOnPostAfterSuccess
{
    public static void Refused(Pipeline<IntakeView> pipeline)
        => pipeline.Post<IntakeView, IntakeRequest>("/Wards/{ward}/Intake", request => request.Form("intake"))
            .OnSuccess(Content.Json<IntakeCreated>(), (created, done) => done.Notice().Success("Resident admitted"))
            .Retry(3, TimeSpan.FromSeconds(2));
}
