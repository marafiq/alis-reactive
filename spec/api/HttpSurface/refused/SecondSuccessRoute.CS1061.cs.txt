using System;
using Alis.Reactive.Elements;
using Alis.Reactive.Pipelines;
using Alis.Reactive.Requests;

namespace Intake;

public static class SecondSuccessRoute
{
    public static void Refused(Pipeline<IntakeView> pipeline, RequestForm<IntakeRequest> intake)
        => pipeline.Post<IntakeView, IntakeRequest>("/Wards/{ward}/Intake", request => request.Form(intake))
            .OnSuccess<IntakeCreated>((created, done) => done.Notice().Success("Resident admitted"))
            .OnSuccess(Content.None, (again, done) => done.Notice().Info("Admitted again"));
}
