using Alis.Reactive.Elements;
using Alis.Reactive.Pipelines;
using Alis.Reactive.Requests;

namespace Intake;

public static class ContainerOfAnotherModel
{
    public static void Refused(Pipeline<IntakeView> pipeline, RequestContainer<BedRequest> beds)
        => pipeline.Post<IntakeView, IntakeRequest>("/Wards/{ward}/Intake", request => request
            .Json(beds, body => body.Set(r => r.Name, "Ann")));
}
