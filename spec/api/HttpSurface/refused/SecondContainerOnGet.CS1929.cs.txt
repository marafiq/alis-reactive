using Alis.Reactive.Elements;
using Alis.Reactive.Pipelines;
using Alis.Reactive.Requests;

namespace Intake;

public static class SecondContainerOnGet
{
    public static void Refused(Pipeline<IntakeView> pipeline, RequestContainer<BedQuery> search, RequestContainer<BedQuery> filters)
        => pipeline.Get<BedQuery>("/Wards/{ward}/Beds", query => query.Inputs(search).Inputs(filters));
}
