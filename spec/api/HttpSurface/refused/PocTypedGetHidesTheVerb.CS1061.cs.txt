using System;
using Alis.Reactive.Elements;
using Alis.Reactive.Pipelines;
using Alis.Reactive.Requests;

namespace Intake;

public static class PocTypedGetHidesTheVerb
{
    public static void Refused(Pipeline<IntakeView> next)
        => next.Get<SummaryQuery>("/Residents/{residentId}/Summary", query => query
                .Set(q => q.ResidentId, 12))
            .Into("resident-summary");
}
