using System;
using Alis.Reactive.Elements;
using Alis.Reactive.Pipelines;
using Alis.Reactive.Requests;

namespace Intake;

public static class ContentReaderOnHeadError
{
    public static void Refused(Pipeline<IntakeView> pipeline)
        => pipeline.Head<IntakeView, DocumentQuery>("/Documents/{id}", query => query.Route(d => d.Id, 7))
            .OnError(404, Content.Text, (missing, shown) => shown.Notice().Warning(missing.Body));
}
